RBC Wealth Advisor Copilot – Structured Client Portfolio Analysis

Demonstrates a structured LCEL chain that ingests a real client profile from RBC's CRM, runs it through GPT-4o, and returns a Pydantic-validated portfolio recommendation — including risk profile, product suggestions, rebalancing actions, and compliance flags — ready for an advisor to review before acting.

from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import PydanticOutputParser
from pydantic import BaseModel, Field
from typing import Literal

# --- Structured output schema for portfolio recommendations ---
class PortfolioRecommendation(BaseModel):
    risk_profile: Literal["Conservative", "Balanced", "Growth", "Aggressive"] = Field(
        description="Client's assessed risk profile"
    )
    recommended_products: list[str] = Field(
        description="List of RBC product names suited to the client"
    )
    rebalancing_actions: list[str] = Field(
        description="Specific rebalancing steps the advisor should discuss with the client"
    )
    rationale: str = Field(
        description="Plain-language explanation of the recommendation for the advisor"
    )
    compliance_flags: list[str] = Field(
        description="Any suitability or regulatory considerations to review before proceeding"
    )

# --- Client context (would be pulled from RBC CRM / core banking in production) ---
client_profile = {
    "name": "Margaret Okonkwo",
    "age": 54,
    "annual_income": 210_000,
    "investment_horizon_years": 11,
    "current_holdings": "65% equities (heavy tech), 30% fixed income, 5% cash",
    "stated_goal": "Retire at 65 with $3.2M, preserve capital in final 3 years",
    "recent_life_event": "Received $400K inheritance last month",
}

parser = PydanticOutputParser(pydantic_object=PortfolioRecommendation)

prompt = ChatPromptTemplate.from_messages([
    ("system",
     "You are an RBC wealth management AI copilot assisting a certified financial advisor. "
     "Analyse the client profile and provide a structured portfolio recommendation aligned with "
     "RBC InvestEase, RBC Direct Investing, and RBC Dominion Securities product suites. "
     "Always surface compliance flags for advisor review.\n\n{format_instructions}"),
    ("human",
     "Client profile:\n"
     "Name: {name} | Age: {age} | Income: ${annual_income:,}\n"
     "Investment horizon: {investment_horizon_years} years\n"
     "Current holdings: {current_holdings}\n"
     "Goal: {stated_goal}\n"
     "Recent life event: {recent_life_event}\n\n"
     "Provide a structured portfolio recommendation."),
])

llm = ChatOpenAI(model="gpt-4o", temperature=0.2)

# LCEL chain: prompt → LLM → structured Pydantic output
chain = prompt | llm | parser

recommendation: PortfolioRecommendation = chain.invoke({
    **client_profile,
    "format_instructions": parser.get_format_instructions(),
})

print(f"Risk Profile : {recommendation.risk_profile}")
print(f"Products     : {', '.join(recommendation.recommended_products)}")
print(f"Actions      : {recommendation.rebalancing_actions}")
print(f"Rationale    : {recommendation.rationale}")
print(f"⚠ Compliance : {recommendation.compliance_flags}")

RBC Retail Banking Copilot – Multi-Step Conversational Agent with Account Tools

Builds a LangGraph ReAct agent that acts as RBC's NOMI copilot, autonomously calling mock core-banking tools to retrieve account balances, check savings goal progress, and recommend relevant RBC products — all within a single natural-language customer conversation.

from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from langchain_core.messages import HumanMessage
from langgraph.prebuilt import create_react_agent
from datetime import date

# ---------------------------------------------------------------------------
# Mock tools – in production these call RBC's Open Banking / core APIs
# ---------------------------------------------------------------------------
@tool
def get_account_summary(account_id: str) -> dict:
    """Return balance and recent transaction summary for an RBC account."""
    # Simulated response from RBC core banking system
    return {
        "account_id": account_id,
        "type": "RBC Signature No Limit Banking",
        "balance_cad": 4_823.17,
        "available_cad": 4_623.17,
        "last_5_transactions": [
            {"date": "2025-07-08", "desc": "Tim Hortons", "amount": -4.75},
            {"date": "2025-07-07", "desc": "Payroll Deposit – Maple Corp", "amount": 3_200.00},
            {"date": "2025-07-06", "desc": "RBC Visa Payment", "amount": -850.00},
            {"date": "2025-07-05", "desc": "Sobeys Grocery", "amount": -112.34},
            {"date": "2025-07-04", "desc": "Interac e-Transfer – Lena", "amount": -200.00},
        ],
    }

@tool
def get_savings_goal_progress(goal_name: str) -> dict:
    """Retrieve progress toward a named RBC NOMI savings goal."""
    goals = {
        "Emergency Fund": {"target": 10_000, "saved": 6_750, "monthly_contribution": 500},
        "Home Down Payment": {"target": 80_000, "saved": 22_400, "monthly_contribution": 1_200},
    }
    return goals.get(goal_name, {"error": f"No goal named '{goal_name}' found."})

@tool
def recommend_rbc_product(client_need: str) -> str:
    """Suggest the most relevant RBC product for a stated client need."""
    need_map = {
        "high interest savings": "RBC High Interest eSavings Account (currently 4.50% promotional rate)",
        "first home": "RBC First Home Savings Account (FHSA) – tax-free contributions up to $8,000/yr",
        "travel rewards": "RBC Avion Visa Infinite – up to 1.25 pts per $1, comprehensive travel insurance",
        "mortgage": "RBC Homeline Plan – flexible HELOC + mortgage combo",
    }
    for keyword, product in need_map.items():
        if keyword in client_need.lower():
            return product
    return "Please speak with an RBC advisor for personalised product guidance."

# ---------------------------------------------------------------------------
# Agent setup using LangGraph's prebuilt ReAct agent
# ---------------------------------------------------------------------------
llm = ChatOpenAI(model="gpt-4o", temperature=0)

rbc_tools = [get_account_summary, get_savings_goal_progress, recommend_rbc_product]

agent = create_react_agent(
    model=llm,
    tools=rbc_tools,
    prompt=(
        "You are NOMI, RBC's friendly and knowledgeable banking copilot. "
        "Help clients with account inquiries, savings goals, and product recommendations. "
        f"Today's date is {date.today()}. Always confirm figures before suggesting actions."
    ),
)

# ---------------------------------------------------------------------------
# Simulate a self-serve retail customer session
# ---------------------------------------------------------------------------
user_query = (
    "Hi NOMI! Can you check my account ACC-00291847 and tell me how my Emergency Fund "
    "goal is going? Also, I'm thinking of buying my first home in 3 years – any RBC products "
    "that could help me save faster?"
)

response = agent.invoke({"messages": [HumanMessage(content=user_query)]})
print(response["messages"][-1].content)
