Priceline Multi-Step Itinerary Change Agent

A fully operational LangChain tool-calling agent that handles a realistic Priceline customer request: looking up a booking, searching alternative flights, applying a Best Price Guarantee match, and confirming the itinerary change — all in a single conversational turn.

from langchain_openai import ChatOpenAI
from langchain.agents import AgentExecutor, create_tool_calling_agent
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder
from langchain_core.tools import tool
from pydantic import BaseModel
from typing import Literal

# --- Priceline Domain Tools ---

@tool
def lookup_booking(booking_ref: str) -> dict:
    """Retrieve flight, hotel, or rental car booking details by reference number."""
    # Simulated booking database response
    return {
        "ref": booking_ref,
        "type": "flight",
        "passenger": "Jane Doe",
        "route": "JFK → LAX",
        "departure": "2025-08-10T08:00",
        "fare_class": "economy",
        "price_paid": 312.50,
        "changeable": True,
        "cancellable": True,
    }

@tool
def search_alternative_flights(origin: str, destination: str, date: str) -> list[dict]:
    """Search available flights for rebooking on a new date."""
    return [
        {"flight": "AA204", "departure": f"{date}T06:00", "price": 289.00, "seats": 4},
        {"flight": "DL887", "departure": f"{date}T11:30", "price": 341.00, "seats": 9},
        {"flight": "UA512", "departure": f"{date}T18:45", "price": 265.00, "seats": 2},
    ]

@tool
def apply_price_match(booking_ref: str, competitor_price: float, competitor_name: str) -> dict:
    """Apply Priceline's Best Price Guarantee if a lower price is found elsewhere."""
    savings = max(0, 312.50 - competitor_price)
    return {
        "approved": savings > 0,
        "original_price": 312.50,
        "matched_price": competitor_price,
        "credit_issued": round(savings, 2),
        "message": f"Price match vs {competitor_name} approved. ${savings:.2f} credit applied.",
    }

@tool
def modify_booking(booking_ref: str, new_flight: str, new_departure: str) -> dict:
    """Confirm a flight change and issue an updated itinerary."""
    return {
        "status": "confirmed",
        "new_flight": new_flight,
        "new_departure": new_departure,
        "confirmation_email": "jane.doe@email.com",
        "change_fee": 0.00,  # waived for elite members
    }

# --- Agent Setup ---

llm = ChatOpenAI(model="gpt-4o", temperature=0)
tools = [lookup_booking, search_alternative_flights, apply_price_match, modify_booking]

prompt = ChatPromptTemplate.from_messages([
    ("system",
     "You are a Priceline travel support agent. Help customers manage bookings, "
     "find alternatives, apply price matches, and confirm changes. "
     "Always look up the booking first, then guide the customer step by step. "
     "Be concise, friendly, and proactive about saving the customer money."),
    MessagesPlaceholder("chat_history", optional=True),
    ("human", "{input}"),
    MessagesPlaceholder("agent_scratchpad"),
])

agent = create_tool_calling_agent(llm, tools, prompt)
agent_executor = AgentExecutor(agent=agent, tools=tools, verbose=True)

# --- Example: Customer wants to change flight and check price match ---
response = agent_executor.invoke({
    "input": (
        "Hi, my booking ref is PCL-88421. I need to move my flight to August 12th. "
        "Also, I saw the same JFK-LAX flight on Expedia for $265 — can you match that?"
    )
})
print(response["output"])

Priceline Structured Cancellation & Refund Classifier

An LCEL chain that uses structured Pydantic output to power Priceline's cancellation and refund decision engine — parsing booking context, applying policy rules (free cancellation windows, carrier schedule changes, VIP exceptions), and returning a fully typed refund decision with a customer-ready message.

from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import PydanticOutputParser
from pydantic import BaseModel, Field
from typing import Literal

# --- Structured Output Schema for Cancellation Decisions ---

class CancellationDecision(BaseModel):
    """Structured output for Priceline's cancellation & refund policy engine."""
    booking_type: Literal["flight", "hotel", "rental_car"] = Field(
        description="Type of travel booking being cancelled"
    )
    refund_eligible: bool = Field(
        description="Whether the customer qualifies for a refund under Priceline policy"
    )
    refund_amount: float = Field(
        description="Dollar amount to refund; 0.0 if not eligible"
    )
    refund_method: Literal["original_payment", "priceline_credit", "none"] = Field(
        description="How the refund will be issued"
    )
    cancellation_fee: float = Field(
        description="Any applicable cancellation fee deducted from refund"
    )
    reason_code: Literal[
        "within_free_cancellation_window",
        "non_refundable_fare",
        "schedule_change_by_carrier",
        "medical_emergency",
        "vip_goodwill_exception",
    ] = Field(description="Policy reason driving the refund decision")
    customer_message: str = Field(
        description="Friendly, plain-English explanation to send to the customer"
    )

# --- LCEL Chain ---

parser = PydanticOutputParser(pydantic_object=CancellationDecision)
llm = ChatOpenAI(model="gpt-4o", temperature=0)

prompt = ChatPromptTemplate.from_messages([
    ("system",
     "You are Priceline's cancellation policy engine. Analyze the booking context "
     "and customer request, then return a structured refund decision. "
     "Apply Priceline's 24-hour free cancellation rule for flights, honor carrier-initiated "
     "schedule changes with full refunds, and use goodwill exceptions for VIP customers. "
     "\n\n{format_instructions}"),
    ("human", "{cancellation_request}"),
]).partial(format_instructions=parser.get_format_instructions())

cancellation_chain = prompt | llm | parser

# --- Example Requests ---

requests = [
    # Scenario 1: Non-refundable fare, but carrier changed the schedule
    {
        "cancellation_request": (
            "Booking PCL-77302: Non-refundable economy flight ORD→MIA, $198 paid. "
            "Airline moved departure from 7am to 11:30pm — customer wants to cancel."
        )
    },
    # Scenario 2: Hotel cancellation within free window by a VIP member
    {
        "cancellation_request": (
            "Booking PCL-55910: Hotel reservation at Marriott Miami, $420, check-in in 3 days. "
            "Customer is Priceline VIP Gold. Hotel policy allows free cancellation up to 48hrs prior. "
            "Customer is requesting cancellation now."
        )
    },
]

for req in requests:
    decision: CancellationDecision = cancellation_chain.invoke(req)
    print(f"Refund Eligible : {decision.refund_eligible}")
    print(f"Refund Amount   : ${decision.refund_amount:.2f} via {decision.refund_method}")
    print(f"Reason Code     : {decision.reason_code}")
    print(f"Customer Message: {decision.customer_message}")
    print("-" * 60)
