AI Clinical Documentation Generator python
Converts a raw provider encounter transcript into a structured SOAP note with ICD-10 and CPT billing codes using LangChain's structured output and LCEL pipeline, ready for EHR integration.
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import JsonOutputParser
from pydantic import BaseModel, Field
from typing import Optional

# Structured output schema for a clinical SOAP note
class SOAPNote(BaseModel):
    subjective: str = Field(description="Patient-reported symptoms and chief complaint")
    objective: str = Field(description="Clinician observations, vitals, and exam findings")
    assessment: str = Field(description="Diagnosis or differential diagnoses")
    plan: str = Field(description="Treatment plan, medications, and follow-up")
    icd10_codes: list[str] = Field(description="Relevant ICD-10 diagnosis codes")
    cpt_codes: list[str] = Field(description="Relevant CPT procedure codes for billing")

# Raw encounter transcript from provider dictation or EHR input
encounter_transcript = """
Patient is a 58-year-old male presenting with a 3-week history of persistent dry cough,
shortness of breath on exertion, and mild chest tightness. No fever or chills. 
History of hypertension managed with lisinopril. BP 138/88, HR 82, SpO2 97% on room air.
Lung auscultation reveals mild expiratory wheezing bilaterally. CXR ordered, 
prescribing albuterol inhaler PRN and scheduling pulmonology referral.
"""

llm = ChatOpenAI(model="gpt-4o", temperature=0)

prompt = ChatPromptTemplate.from_messages([
    ("system",
     "You are a clinical documentation specialist for an athenahealth-integrated EHR. "
     "Convert provider encounter notes into structured SOAP format with accurate "
     "ICD-10 and CPT codes for billing compliance."),
    ("human", "Encounter transcript:\n{transcript}\n\nGenerate a structured SOAP note with billing codes.")
])

# LCEL chain: prompt -> LLM -> structured JSON output
documentation_chain = prompt | llm.with_structured_output(SOAPNote)

soap_note = documentation_chain.invoke({"transcript": encounter_transcript})

print("=== Generated SOAP Note ===")
print(f"S: {soap_note.subjective}")
print(f"O: {soap_note.objective}")
print(f"A: {soap_note.assessment}")
print(f"P: {soap_note.plan}")
print(f"ICD-10 Codes: {', '.join(soap_note.icd10_codes)}")
print(f"CPT Codes:    {', '.join(soap_note.cpt_codes)}")
Prior Authorization Decision Support Agent python
An agentic prior authorization workflow that uses tool-calling to check payer policies, validate clinical necessity criteria, and auto-draft PA submission letters, reducing administrative burden on clinical staff.
from langchain_anthropic import ChatAnthropic
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.tools import tool
from langchain.agents import AgentExecutor, create_tool_calling_agent
from pydantic import BaseModel, Field

# Simulated payer policy and formulary lookups (would call athenahealth APIs in production)
@tool
def lookup_payer_policy(payer_id: str, procedure_code: str) -> str:
    """Look up a payer's prior authorization requirements for a given procedure code."""
    mock_policies = {
        ("BCBS-MA", "27447"): "PA required. Must document conservative therapy (PT x6 weeks) and X-ray evidence of OA.",
        ("AETNA", "70553"): "PA required. Indicate clinical indication and prior imaging results.",
    }
    return mock_policies.get((payer_id, procedure_code), "No PA required per current payer policy.")

@tool
def check_clinical_criteria(diagnosis_code: str, procedure_code: str) -> str:
    """Check evidence-based clinical criteria and medical necessity guidelines for a procedure."""
    mock_criteria = {
        ("M17.11", "27447"): "Criteria met if: BMI < 40, conservative treatment failed, functional limitation documented.",
    }
    return mock_criteria.get((diagnosis_code, procedure_code), "Consult payer LCD/NCD guidelines for specific criteria.")

@tool
def draft_pa_letter(patient_name: str, procedure: str, clinical_summary: str) -> str:
    """Draft a prior authorization request letter for submission to the payer."""
    return (
        f"Dear Medical Director,\n\nWe are requesting prior authorization for {procedure} "
        f"for patient {patient_name}. Clinical summary: {clinical_summary}\n\n"
        f"Please find supporting documentation attached.\n\nSincerely,\nProvider"
    )

tools = [lookup_payer_policy, check_clinical_criteria, draft_pa_letter]

llm = ChatAnthropic(model="claude-3-5-sonnet-20241022", temperature=0)

prompt = ChatPromptTemplate.from_messages([
    ("system",
     "You are a prior authorization specialist at an athenahealth-powered medical practice. "
     "Use the available tools to assess PA requirements, verify clinical criteria, "
     "and draft submission letters to minimize claim denials."),
    ("human", "{input}"),
    ("placeholder", "{agent_scratchpad}"),
])

agent = create_tool_calling_agent(llm, tools, prompt)
pa_agent = AgentExecutor(agent=agent, tools=tools, verbose=True)

# Realistic PA request scenario
pa_request = (
    "Patient: Jane Doe | Payer: BCBS-MA | Procedure: 27447 (Total Knee Arthroplasty) | "
    "Diagnosis: M17.11 (Primary OA, right knee). She has completed 8 weeks of PT with "
    "no improvement and X-rays confirm severe joint space narrowing. Please assess PA "
    "requirements, verify she meets clinical criteria, and draft a PA request letter."
)

result = pa_agent.invoke({"input": pa_request})
print("\n=== PA Agent Output ===")
print(result["output"])
