FanDuel Bet Inquiry & Dispute Resolution Agent python
A tool-calling LangChain agent that handles FanDuel bet disputes end-to-end — fetching bet records, reviewing account standing, and autonomously issuing refunds or escalating to a human agent based on configurable business rules.
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.tools import tool
from langchain.agents import AgentExecutor, create_tool_calling_agent
from pydantic import BaseModel, Field
from typing import Literal

# --- Structured output schema for dispute resolution decisions ---
class DisputeResolution(BaseModel):
    bet_id: str = Field(description="The bet ID under dispute")
    decision: Literal["approved", "denied", "escalate_to_human"] = Field(description="Resolution outcome")
    refund_amount: float = Field(description="Refund amount in USD, 0.0 if denied")
    explanation: str = Field(description="Customer-facing explanation of the decision")

# --- Mock tools representing FanDuel's internal APIs ---
@tool
def get_bet_details(bet_id: str) -> dict:
    """Fetch bet details from FanDuel's bet ledger by bet ID."""
    # Simulated bet record
    return {
        "bet_id": bet_id, "user_id": "FD-8821903", "sport": "NFL",
        "market": "Kansas City Chiefs ML", "stake": 50.00, "odds": -110,
        "status": "settled_loss", "settled_at": "2024-01-14T23:05:00Z",
        "dispute_eligible": True, "game_result": "Chiefs won 27-20"
    }

@tool
def get_account_history(user_id: str) -> dict:
    """Retrieve a customer's account standing and dispute history."""
    return {"user_id": user_id, "account_status": "good_standing",
            "prior_disputes": 0, "total_deposits": 1200.00, "win_rate": 0.44}

@tool
def submit_refund(bet_id: str, amount: float, reason: str) -> str:
    """Process a refund to the customer's FanDuel wallet."""
    return f"Refund of ${amount:.2f} for bet {bet_id} submitted successfully. Reason: {reason}"

llm = ChatOpenAI(model="gpt-4o", temperature=0)
tools = [get_bet_details, get_account_history, submit_refund]

prompt = ChatPromptTemplate.from_messages([
    ("system", """You are a FanDuel support agent specializing in bet disputes.
    Always fetch bet details and account history before making a resolution decision.
    Approve refunds only when there is a clear data discrepancy (e.g., bet marked lost
    despite the customer's selection winning). Escalate repeat disputes or large stakes (>$500) to a human."""),
    ("human", "{input}"),
    ("placeholder", "{agent_scratchpad}"),
])

agent = create_tool_calling_agent(llm, tools, prompt)
agent_executor = AgentExecutor(agent=agent, tools=tools, verbose=True)

# --- Simulate an incoming customer dispute ---
customer_message = (
    "Hi, my bet ID is BET-449921. I placed $50 on the Chiefs moneyline and they WON, "
    "but my account shows a loss. Can you look into this and refund me?"
)

response = agent_executor.invoke({"input": customer_message})
print("\n=== Agent Response ===")
print(response["output"])
FanDuel Responsible Gambling Intervention & Deposit Triage Chain python
An LCEL chain that triages inbound FanDuel support tickets by enriching them with live CRM data, then uses Claude to classify urgency, detect responsible gambling risk signals, and generate a compliant, empathetic draft response — all as a validated Pydantic object.
from langchain_anthropic import ChatAnthropic
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import PydanticOutputParser
from langchain_core.runnables import RunnableLambda, RunnablePassthrough
from pydantic import BaseModel, Field
from typing import Literal

# --- Structured triage schema for support tickets ---
class SupportTriage(BaseModel):
    ticket_type: Literal["deposit_issue", "withdrawal_issue", "responsible_gambling", "general_inquiry"]
    urgency: Literal["low", "medium", "high", "critical"] = Field(
        description="critical = immediate RG intervention required")
    recommended_action: str = Field(description="Specific next step for the support agent or system")
    rg_intervention_required: bool = Field(description="True if responsible gambling resources must be surfaced")
    draft_response: str = Field(description="Empathetic, compliant draft reply to send to the customer")

# --- Simulated customer context loader (would query FanDuel's CRM in production) ---
def load_customer_context(inputs: dict) -> dict:
    """Enrich the ticket with account data from FanDuel's CRM."""
    return {
        **inputs,
        "customer_context": {
            "user_id": "FD-3341087",
            "deposit_count_last_24h": 7,       # High-frequency deposits — RG signal
            "net_loss_last_7d": -843.00,
            "withdrawal_pending": True,
            "withdrawal_amount": 200.00,
            "account_status": "active",
            "self_exclusion_history": False,
        }
    }

llm = ChatAnthropic(model="claude-3-5-sonnet-20241022", temperature=0)
parser = PydanticOutputParser(pydantic_object=SupportTriage)

prompt = ChatPromptTemplate.from_messages([
    ("system", """You are a FanDuel customer support triage specialist trained in responsible gambling (RG) protocols.

Analyze the customer message AND their account context to produce a structured triage decision.
RG intervention is CRITICAL if: deposit_count_last_24h > 5, net_loss_last_7d < -500, or the customer
expresses distress. Always draft empathetic, jargon-free responses. Comply with AGA Safe Play guidelines.

{format_instructions}"""),
    ("human", "Customer message: {message}\n\nAccount context: {customer_context}"),
])

# --- LCEL chain: enrich context → format prompt → call LLM → parse structured output ---
triage_chain = (
    RunnablePassthrough()                          # Pass raw input through
    | RunnableLambda(load_customer_context)        # Enrich with CRM data
    | prompt.partial(format_instructions=parser.get_format_instructions())
    | llm
    | parser
)

# --- Simulate an inbound support message ---
ticket = {
    "message": (
        "I've been trying to withdraw my $200 for 3 days and it's still pending. "
        "I'm really stressed out, I need that money back. Please help."
    )
}

triage_result: SupportTriage = triage_chain.invoke(ticket)

print(f"Ticket Type   : {triage_result.ticket_type}")
print(f"Urgency       : {triage_result.urgency}")
print(f"RG Intervention: {triage_result.rg_intervention_required}")
print(f"Action        : {triage_result.recommended_action}")
print(f"\nDraft Response:\n{triage_result.draft_response}")
