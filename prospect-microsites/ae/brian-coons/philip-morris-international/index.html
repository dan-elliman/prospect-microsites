Regulatory Filing Section Synthesizer

Uses an LCEL chain with structured Pydantic output to automatically synthesize aerosol chemistry study data into a formatted, compliance-assessed PMTA section for FDA submission — complete with scientific claims, risk disclosures, and recommended next steps for the regulatory affairs team.

from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import JsonOutputParser
from pydantic import BaseModel, Field
from typing import Literal

# --- Structured output schema for a regulatory filing section ---
class RegulatorySection(BaseModel):
    section_title: str = Field(description="Official section title per FDA PMTA format")
    summary: str = Field(description="Executive summary of the section (2-3 sentences)")
    scientific_claims: list[str] = Field(description="List of evidence-backed scientific claims")
    risk_disclosures: list[str] = Field(description="Required risk disclosures for this section")
    compliance_status: Literal["compliant", "needs_review", "non_compliant"] = Field(
        description="Assessed compliance status based on provided data"
    )
    recommended_actions: list[str] = Field(description="Next steps for the regulatory affairs team")

# --- PMI product context ---
PRODUCT_NAME = "IQOS ILUMA"
SUBMISSION_TYPE = "Premarket Tobacco Product Application (PMTA)"
TARGET_AUTHORITY = "U.S. Food and Drug Administration (FDA)"

llm = ChatOpenAI(model="gpt-4o", temperature=0)
parser = JsonOutputParser(pydantic_object=RegulatorySection)

prompt = ChatPromptTemplate.from_messages([
    ("system",
     "You are a regulatory affairs specialist at Philip Morris International. "
     "Your role is to synthesize scientific study data into structured sections "
     "for {submission_type} submissions to {authority}. "
     "Be precise, cite evidence, and flag any compliance gaps.\n\n"
     "{format_instructions}"),
    ("human",
     "Product: {product_name}\n"
     "Section to draft: {section_name}\n"
     "Study data summary:\n{study_data}")
])

# --- LCEL chain ---
chain = prompt | llm | parser

# --- Example invocation with aerosol chemistry study data ---
study_data = """
- IQOS ILUMA aerosol contains 95% fewer harmful chemicals vs. cigarette smoke (PMI internal study, 2022)
- No combustion occurs; peak aerosol temp: 350°C vs. 600°C for cigarettes
- 12-month clinical study: statistically significant reduction in BoExposure markers in adult switchers
- Nicotine delivery profile comparable to combustible cigarettes (bioequivalence criterion met)
"""

result = chain.invoke({
    "submission_type": SUBMISSION_TYPE,
    "authority": TARGET_AUTHORITY,
    "product_name": PRODUCT_NAME,
    "section_name": "Section 4 – Aerosol Chemistry and Toxicological Assessment",
    "study_data": study_data,
    "format_instructions": parser.get_format_instructions()
})

print(f"Section: {result['section_title']}")
print(f"Compliance Status: {result['compliance_status'].upper()}")
print(f"\nSummary:\n{result['summary']}")
print(f"\nScientific Claims:")
for claim in result['scientific_claims']:
    print(f"  • {claim}")
print(f"\nRecommended Actions:")
for action in result['recommended_actions']:
    print(f"  → {action}")

Multi-Market Regulatory Gap Analysis Agent

Implements a tool-calling LangChain agent powered by Claude that autonomously fetches market-specific regulatory requirements (Japan MHLW, EU TPD2, Health Canada) and cross-checks them against PMI's product dossier content, surfacing actionable compliance gaps before submission deadlines.

from langchain_anthropic import ChatAnthropic
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.tools import tool
from langchain.agents import AgentExecutor, create_tool_calling_agent
from pydantic import BaseModel, Field

# --- Tool input schemas ---
class MarketLookupInput(BaseModel):
    market: str = Field(description="Country or region name, e.g. 'Japan', 'EU', 'Canada'")
    product_category: str = Field(description="Product category: 'HTP', 'e-vapor', 'oral_nicotine'")

class ComplianceCheckInput(BaseModel):
    regulation_text: str = Field(description="Regulatory requirement text to check")
    dossier_section: str = Field(description="Relevant PMI dossier section content")

# --- Simulated regulatory knowledge base tools ---
@tool(args_schema=MarketLookupInput)
def fetch_market_regulations(market: str, product_category: str) -> str:
    """Retrieves current regulatory requirements for a PMI product category in a given market."""
    # In production: query a vector store or regulatory database (e.g., Pinecone + WHO/FDA docs)
    mock_db = {
        ("Japan", "HTP"): "Japan MHLW requires toxicological equivalence data and Japanese-language labeling. "
                          "Nicotine content must be declared per stick. Annual post-market surveillance reports mandatory.",
        ("EU", "HTP"): "EU TPD2 Article 19 requires 6-month pre-notification to national authorities. "
                       "Emissions testing per ISO 20778 mandatory. Health warnings must cover 65% of packaging.",
        ("Canada", "oral_nicotine"): "Health Canada requires pre-market notification under TVPA. "
                                     "Max nicotine concentration 4mg/g. French/English bilingual labeling required."
    }
    return mock_db.get((market, product_category),
                       f"No cached regulation found for {product_category} in {market}. Escalate to local RA team.")

@tool(args_schema=ComplianceCheckInput)
def check_dossier_compliance(regulation_text: str, dossier_section: str) -> str:
    """Compares a regulatory requirement against the current PMI submission dossier section and returns a gap analysis."""
    # In production: use semantic similarity + LLM critique chain
    if "ISO 20778" in regulation_text and "ISO 20778" not in dossier_section:
        return "GAP IDENTIFIED: Dossier is missing ISO 20778 emissions testing data. Action required before submission."
    if "bilingual" in regulation_text and "French" not in dossier_section:
        return "GAP IDENTIFIED: French-language labeling content absent from dossier. Localization team must be engaged."
    return "No critical gaps detected. Section appears aligned with stated regulatory requirement."

# --- Agent setup ---
llm = ChatAnthropic(model="claude-3-5-sonnet-20241022", temperature=0)
tools = [fetch_market_regulations, check_dossier_compliance]

prompt = ChatPromptTemplate.from_messages([
    ("system",
     "You are a senior regulatory affairs agent at Philip Morris International. "
     "Your goal is to identify compliance gaps in PMI product dossiers across multiple markets "
     "before submission to health authorities. Be methodical: look up regulations first, "
     "then check the dossier against each requirement. Summarize all gaps and priorities clearly."),
    ("human", "{input}"),
    ("placeholder", "{agent_scratchpad}")
])

agent = create_tool_calling_agent(llm, tools, prompt)
executor = AgentExecutor(agent=agent, tools=tools, verbose=True)

# --- Run gap analysis for ZYN Canada launch ---
zyn_dossier_excerpt = """
ZYN Nicotine Pouches – Section 3: Labeling & Packaging
Nicotine concentration: 3mg per pouch. English-language health warnings cover 65% of front panel.
Product is tobacco-free. Intended for adult nicotine users only.
"""

result = executor.invoke({
    "input": f"Analyze our ZYN oral nicotine dossier for Canada market readiness. "
             f"Dossier excerpt: {zyn_dossier_excerpt}"
})

print("\n=== REGULATORY GAP ANALYSIS REPORT ===")
print(result["output"])
